# --- FUNKCJE POMOCNICZE ---

def load_csv(filename):
    """Ładuje prosty plik CSV (dla tabeli i strzelców)."""
    if os.path.exists(filename):
        return pd.read_csv(filename)
    else:
        st.error(f"Nie znaleziono pliku: {filename}")
        return None

def parse_team_file(filename):
    """
    Parsuje złożony plik zespołu, który zawiera wiele sekcji (Skład, Sztab, Mecze)
    w jednym pliku CSV.
    """
    if not os.path.exists(filename):
        st.error(f"Brak danych szczegółowych dla tego zespołu (plik {filename} nie istnieje).")
        return None, None, None

    with open(filename, 'r', encoding='utf-8') as f:
        lines = f.readlines()

    # Znajdowanie punktów podziału w pliku
    squad_start = 0
    staff_start = -1
    matches_start = -1

    for i, line in enumerate(lines):
        if "numer,imię i nazwisko" in line and i == 0:
            squad_start = i
        elif "lp,imię i nazwisko,funkcja" in line or "rola,narodowość" in line:
            staff_start = i
        elif "kolejka,gospodarze,goście" in line:
            matches_start = i

    # Wczytywanie sekcji SKŁAD
    squad_data = ""
    end_squad = staff_start if staff_start != -1 else (matches_start if matches_start != -1 else len(lines))
    
    # Pomijamy wiersze podsumowujące (często są puste lub zawierają tylko statystyki zbiorcze przed nową sekcją)
    raw_squad_lines = lines[squad_start:end_squad]
    clean_squad_lines = [line for line in raw_squad_lines if line.strip() and not line.startswith(',,,,,,,')]
    squad_data = "".join(clean_squad_lines)
    
    df_squad = pd.read_csv(io.StringIO(squad_data))

    # Wczytywanie sekcji SZTAB (jeśli istnieje)
    df_staff = pd.DataFrame()
    if staff_start != -1:
        end_staff = matches_start if matches_start != -1 else len(lines)
        raw_staff_lines = lines[staff_start:end_staff]
        clean_staff_lines = [line for line in raw_staff_lines if line.strip() and not line.startswith(',,,,,,,')]
        staff_data = "".join(clean_staff_lines)
        try:
            df_staff = pd.read_csv(io.StringIO(staff_data))
        except:
            pass # Ignoruj błędy parsowania sztabu

    # Wczytywanie sekcji MECZE (jeśli istnieje)
    df_matches = pd.DataFrame()
    if matches_start != -1:
        matches_data = "".join(lines[matches_start:])
        try:
            df_matches = pd.read_csv(io.StringIO(matches_data))
        except:
            pass

    return df_squad, df_staff, df_matches
